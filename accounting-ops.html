⚙️ 建立 SOP 連結資料表

只需執行一次。到 Supabase → SQL Editor 貼上並執行以下指令，之後每個平台卡片就能新增／編輯 SOP 連結：

CREATE TABLE IF NOT EXISTS sop_links (
  key text PRIMARY KEY,
  url text NOT NULL,
  updated_at timestamptz DEFAULT now()
);
ALTER TABLE sop_links ENABLE ROW LEVEL SECURITY;
-- 本頁（accounting-ops.html）不需登入，因此開放 anon 角色讀寫
DO $$ BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename='sop_links' AND policyname='public_select') THEN
    CREATE POLICY "public_select" ON sop_links FOR SELECT TO anon, authenticated USING (true);
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename='sop_links' AND policyname='public_write') THEN
    CREATE POLICY "public_write" ON sop_links FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);
  END IF;
END $$;
